cmake_minimum_required(VERSION 3.15)

# ─── Options ───────────────────────────────────────────────────────────────
#
# These control whether the library is built as a standalone project or as
# a subproject of a parent build.
#
option(LUCID_FORMAT_IS_SUBPROJECT
    "Build as a subproject of a parent CMake project" OFF)
option(LUCID_FORMAT_BUILD_CLI
    "Build the lucid CLI executable" ON)
option(LUCID_FORMAT_BUILD_TESTS
    "Build the test suite" ON)

# ─── Top-level-only setup ──────────────────────────────────────────────────
#
# The compiler pin, environment checks, and global settings are only
# appropriate when the library is the root project. As a subproject,
# the parent owns the compiler and the build settings.

if(NOT LUCID_FORMAT_IS_SUBPROJECT)

    # Compiler pin (Windows: force MSVC).
    if(WIN32 AND NOT DEFINED CMAKE_CXX_COMPILER)
        file(GLOB MSVC_CL_PATHS
            "C:/Program Files/Microsoft Visual Studio/2022/*/VC/Tools/MSVC/*/bin/Hostx64/x64/cl.exe"
            "C:/Program Files (x86)/Microsoft Visual Studio/2022/*/VC/Tools/MSVC/*/bin/Hostx64/x64/cl.exe"
        )
        if(MSVC_CL_PATHS)
            list(GET MSVC_CL_PATHS 0 MSVC_CL_PATH)
            set(CMAKE_CXX_COMPILER "${MSVC_CL_PATH}" CACHE STRING "C++ compiler" FORCE)
            set(CMAKE_C_COMPILER   "${MSVC_CL_PATH}" CACHE STRING "C compiler"   FORCE)
            message(STATUS "Pinned compiler: ${MSVC_CL_PATH}")
        else()
            message(FATAL_ERROR
                "MSVC not found under C:/Program Files[/ (x86)]/Microsoft Visual "
                "Studio/2022/*.\n"
                "Install Visual Studio 2022 (or Build Tools) with the 'Desktop "
                "development with C++' workload, then reconfigure.")
        endif()
    endif()

    # Environment check.
    if(WIN32 AND MSVC AND CMAKE_GENERATOR STREQUAL "Ninja" AND NOT DEFINED ENV{INCLUDE})
        message(FATAL_ERROR "...")
    endif()

    # Global settings.
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL" CACHE STRING "" FORCE)
    if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
        set(CMAKE_BUILD_TYPE "Release" CACHE STRING "" FORCE)
    endif()

    # Warning options and shared flags.
    option(LUCID_WERROR "Treat warnings as errors" OFF)
    set(LUCID_CXX_WARNINGS
        $<$<CXX_COMPILER_ID:MSVC>:/W4>
        $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic>
    )
    if(LUCID_WERROR)
        list(APPEND LUCID_CXX_WARNINGS
            $<$<CXX_COMPILER_ID:MSVC>:/WX>
            $<$<CXX_COMPILER_ID:GNU,Clang>:-Werror>
        )
    endif()
    set(LUCID_MSVC_FLAGS /FS /MP)

endif()

# ─── Project declaration ───────────────────────────────────────────────────

project(LucidFormat LANGUAGES CXX)

# ─── Compiler enforcement (safety net) ────────────────────────────────────

if(NOT LUCID_FORMAT_IS_SUBPROJECT AND WIN32 AND NOT MSVC)
    message(FATAL_ERROR
        "This project targets MSVC on Windows.\n"
        "  Detected compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\n"
        "  Compiler path:     ${CMAKE_CXX_COMPILER}\n"
        "\n"
        "The pin above should have selected cl.exe. If you see this message,\n"
        "either the pin failed or you passed a different -DCMAKE_CXX_COMPILER.\n"
        "Open 'Developer PowerShell for VS 2022' from the Start menu and\n"
        "reconfigure, or pass -DCMAKE_CXX_COMPILER=<path to cl.exe>.")
endif()

# ─── Warning options (available in both modes) ─────────────────────────────

if(NOT DEFINED LUCID_CXX_WARNINGS)
    set(LUCID_CXX_WARNINGS
        $<$<CXX_COMPILER_ID:MSVC>:/W4>
        $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic>
    )
endif()

if(NOT DEFINED LUCID_MSVC_FLAGS)
    set(LUCID_MSVC_FLAGS /FS /MP)
endif()

# ─── Library: lucid_core ───────────────────────────────────────────────────

    core/src/core/memory/StringPool.cpp
    core/src/core/diagnostics/Diagnostic.cpp
    core/src/core/Tokens.cpp
)

target_include_directories(lucid_core
    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/core/include"
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/core/src"
)

target_compile_features(lucid_core PUBLIC cxx_std_17)
target_compile_options(lucid_core PRIVATE ${LUCID_CXX_WARNINGS})

    target_compile_options(lucid_core PRIVATE ${LUCID_MSVC_FLAGS})
endif()

# ─── CLI (conditional) ─────────────────────────────────────────────────────

if(LUCID_FORMAT_BUILD_CLI)
    add_library(lucid_cli STATIC
        cli/src/cli/CLIOptions.cpp
        cli/src/cli/Registry.cpp
        cli/src/cli/commands/Format.cpp
        cli/src/cli/commands/Check.cpp
        cli/src/cli/commands/Compile.cpp
    )

    target_include_directories(lucid_cli PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/cli/include"
    )

    target_link_libraries(lucid_cli PUBLIC
        lucid_core
        lucid_parser
        lucid_formatter
        lucid_sema
        lucid_serialization
    )

    target_compile_features(lucid_cli PUBLIC cxx_std_17)
    target_compile_options(lucid_cli PRIVATE ${LUCID_CXX_WARNINGS})
    if(MSVC)
        target_compile_options(lucid_cli PRIVATE ${LUCID_MSVC_FLAGS})
    endif()

    add_executable(lucid
        cli/src/cli/Main.cpp
    )

    target_link_libraries(lucid PRIVATE
        lucid_cli
        lucid_serialization
    )

    target_compile_features(lucid PRIVATE cxx_std_17)
    target_compile_options(lucid PRIVATE ${LUCID_CXX_WARNINGS})
    if(MSVC)
        target_compile_options(lucid PRIVATE ${LUCID_MSVC_FLAGS})
    endif()
