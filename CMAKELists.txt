cmake_minimum_required(VERSION 3.15)

# ─── Compiler pin (must come before project()) ─────────────────────────────
#
# On Windows, force MSVC. CMake's default search picks whatever `c++` is
# first on PATH, which on a plain PowerShell is often a MinGW or Strawberry
# Perl compiler. That compiler does not ship MSVC's headers, does not honor
# /FS /MP, and produces a compile_commands.json that clangd cannot resolve
# against the MSVC system headers.
#
# This block runs before `project()`, which is where CMake finalizes the
# compiler. Setting CMAKE_CXX_COMPILER here (as a CACHE variable) pins it
# for every configure, regardless of the calling shell.

if(WIN32 AND NOT DEFINED CMAKE_CXX_COMPILER)
    file(GLOB MSVC_CL_PATHS
        "C:/Program Files/Microsoft Visual Studio/2022/*/VC/Tools/MSVC/*/bin/Hostx64/x64/cl.exe"
        "C:/Program Files (x86)/Microsoft Visual Studio/2022/*/VC/Tools/MSVC/*/bin/Hostx64/x64/cl.exe"
    )
    if(MSVC_CL_PATHS)
        list(GET MSVC_CL_PATHS 0 MSVC_CL_PATH)
        set(CMAKE_CXX_COMPILER "${MSVC_CL_PATH}" CACHE STRING "C++ compiler" FORCE)
        set(CMAKE_C_COMPILER   "${MSVC_CL_PATH}" CACHE STRING "C compiler"   FORCE)
        message(STATUS "Pinned compiler: ${MSVC_CL_PATH}")
    else()
        message(FATAL_ERROR
            "MSVC not found under C:/Program Files[/ (x86)]/Microsoft Visual "
            "Studio/2022/*.\n"
            "Install Visual Studio 2022 (or Build Tools) with the 'Desktop "
            "development with C++' workload, then reconfigure.")
    endif()
endif()

project(LucidFormat LANGUAGES CXX)

# ─── Compiler enforcement (safety net) ─────────────────────────────────────
#
# On Windows, require MSVC. See the comment above for why.

if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR
        "This project targets MSVC on Windows.\n"
        "  Detected compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\n"
        "  Compiler path:     ${CMAKE_CXX_COMPILER}\n"
        "\n"
        "The pin above should have selected cl.exe. If you see this message,\n"
        "either the pin failed or you passed a different -DCMAKE_CXX_COMPILER.\n"
        "Open 'Developer PowerShell for VS 2022' from the Start menu and\n"
        "reconfigure, or pass -DCMAKE_CXX_COMPILER=<path to cl.exe>.")
endif()

# ─── Environment check ─────────────────────────────────────────────────────

# Only require the developer environment when using Ninja. The Visual
# Studio generator resolves the toolchain through the registry and does
# not need INCLUDE set in the shell.
if(WIN32 AND MSVC AND CMAKE_GENERATOR STREQUAL "Ninja"
        AND NOT DEFINED ENV{INCLUDE})
    message(FATAL_ERROR "...")
endif()

# ─── Standard & compiler identity ──────────────────────────────────────────

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# ─── MSVC runtime consistency ──────────────────────────────────────────────

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL" CACHE STRING "" FORCE)

# ─── Default build type for single-config generators ───────────────────────

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE "Release" CACHE STRING "" FORCE)
endif()

# ─── Single-config convenience for the Visual Studio generator ─────────────
#
# The VS generator is multi-config; without this, every `cmake --build` and
# `ctest` needs `--config Release`. Restricting the configuration types to
# Release lets the build and test commands omit the flag. If you later want
# to build Debug, remove this block or add "Debug" to the list.

# ─── Warning options ───────────────────────────────────────────────────────

option(LUCID_WERROR "Treat warnings as errors" OFF)

set(LUCID_CXX_WARNINGS
    $<$<CXX_COMPILER_ID:MSVC>:/W4>
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic>
)

if(LUCID_WERROR)
    list(APPEND LUCID_CXX_WARNINGS
        $<$<CXX_COMPILER_ID:MSVC>:/WX>
        $<$<CXX_COMPILER_ID:GNU,Clang>:-Werror>
    )
endif()

set(LUCID_MSVC_FLAGS
    /FS        # force sequential PDB writes
    /MP        # multi-processor compilation
)

# ─── Library: lucid_core ───────────────────────────────────────────────────

add_library(lucid_core STATIC
    core/src/core/memory/StringPool.cpp
    core/src/core/diagnostics/Diagnostic.cpp
)

target_include_directories(lucid_core
    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/core/include"
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/core/src"
)

target_compile_options(lucid_core PRIVATE ${LUCID_CXX_WARNINGS})
if(MSVC)
    target_compile_options(lucid_core PRIVATE ${LUCID_MSVC_FLAGS})
endif()

# ─── Library: lucid_parser ─────────────────────────────────────────────────

add_library(lucid_parser STATIC
    parser/src/parser/placeholder.cpp
)

target_include_directories(lucid_parser
    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/parser/include"
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/parser/src"
)

target_link_libraries(lucid_parser PUBLIC lucid_core)

target_compile_options(lucid_parser PRIVATE ${LUCID_CXX_WARNINGS})
if(MSVC)
    target_compile_options(lucid_parser PRIVATE ${LUCID_MSVC_FLAGS})
endif()

# ─── Library: lucid_formatter ──────────────────────────────────────────────

add_library(lucid_formatter STATIC
    formatter/src/formatter/placeholder.cpp
)

target_include_directories(lucid_formatter
    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/formatter/include"
    PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/formatter/src"
)

target_link_libraries(lucid_formatter PUBLIC
    lucid_core
    lucid_parser
)

target_compile_options(lucid_formatter PRIVATE ${LUCID_CXX_WARNINGS})
if(MSVC)
    target_compile_options(lucid_formatter PRIVATE ${LUCID_MSVC_FLAGS})
endif()

# ─── Executable: lucid-fmt ─────────────────────────────────────────────────

add_executable(lucid-fmt
    src/main.cpp
)

# lucid_formatter PUBLIC-links lucid_parser and lucid_core, so listing
# only lucid_formatter is enough.
target_link_libraries(lucid-fmt PRIVATE lucid_formatter)

target_compile_options(lucid-fmt PRIVATE ${LUCID_CXX_WARNINGS})
if(MSVC)
    target_compile_options(lucid-fmt PRIVATE ${LUCID_MSVC_FLAGS})
endif()

enable_testing()
add_subdirectory(tests)
